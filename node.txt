$zip="$env:TEMP\node.zip"
Invoke-WebRequest "https://nodejs.org/dist/v22.23.3/node-v22.23.3-win-x64.zip" -OutFile $zip
Expand-Archive $zip -DestinationPath "$env:USERPROFILE\nodejs" -Force
Remove-Item $zip

$nodePath="$env:USERPROFILE\nodejs\node-v22.23.3-win-x64"
[Environment]::SetEnvironmentVariable("Path", $env:Path + ";$nodePath", "User")
$env:Path += ";$nodePath"
